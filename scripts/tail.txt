
    async function saveDevice() {
      const current = devices.find(x => x.id === editingDeviceId);
      const type = editingBasicOnly && current ? (current.conn_type || 'ssh') : document.getElementById('add-type').value;
      const payload = {
        conn_type: type,
        device_type: document.getElementById('add-device-type').value,
        remark: document.getElementById('add-remark').value.trim(),
        username: editingBasicOnly && current ? (current.username || '') : document.getElementById('add-user').value.trim(),
        password: editingBasicOnly && current ? (current.password || '') : document.getElementById('add-pass').value
      };
      if (editingDeviceId) {
        payload.device_id = editingDeviceId;
      }
      if (editingBasicOnly && editingDeviceId) {
        payload.basic_only = true;
      }

      if (type === 'serial') {
        payload.serial_port = editingBasicOnly && current ? (current.serial_port || '') : document.getElementById('add-serial').value.trim();
        payload.baud = editingBasicOnly && current ? parseInt(current.baud || 115200) : parseInt(document.getElementById('add-baud').value || '115200');
      } else {
        payload.ip = editingBasicOnly && current ? (current.ip || '') : document.getElementById('add-ip').value.trim();
        payload.port = editingBasicOnly && current ? parseInt(current.port || (type === 'telnet' ? 23 : 22)) : parseInt(document.getElementById('add-port').value || (type === 'telnet' ? '23' : '22'));
        if (!payload.ip) { alert('请输入 IP'); return; }
      }

      const r = await fetch('/api/device/add', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      let d;
      try { d = await r.json(); } catch(e) { alert('保存失败：服务器错误'); return; }
      if (!d.success) { alert(d.message || '保存失败'); return; }

      closeModal('modal-device');
      await loadDevices();
      buildTopologyFromDevices();
      await syncTopologyState();
      renderAll();
      // 新增设备后自动采集
      collectDeviceFacts(payload.remark || payload.name);
      addMsg('system', editingDeviceId ? '✅ 设备已更新， 自动采集中设备信息' : '⏳ 自动采集中...');
      editingDeviceId = null;
      editingBasicOnly = false;
    }

    async function structuredDiagnose(dev, type) {
      // 结构化诊断：走 /api/diagnose，不走 LLM，确定性排错
      const typeLabels = {vlan:'VLAN', routing:'路由', connectivity:'连通性'};
      const label = typeLabels[type] || type;
      addMsg('user', `🔍 ${label}诊断: ${dev}`);
      addMsg('system', `⏳ 正在执行${label}结构化诊断...`);
      try {
        const r = await fetch('/api/diagnose', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ device: dev, type: type, symptom: `${label}故障排查` })
        });
        const res = await r.json();
        if (!res.success) {
          addMsg('ai', `❌ 诊断失败: ${res.message||'未知错误'}`);
          return;
        }
        // 格式化诊断结果
        let html = `<div style="background:#1e1b4b;border:1px solid #4f46e5;border-radius:8px;padding:12px;">`;
        html += `<div style="color:#a5b4fc;font-weight:600;margin-bottom:8px;">🔍 ${label}诊断结果</div>`;
        if (res.steps && res.steps.length) {
          for (const s of res.steps) {
            const icon = s.status === 'PASS' ? '✅' : (s.status === 'FAIL' ? '❌' : (s.status === 'WARNING' ? '⚠️' : '⏭️'));
            html += `<div style="margin-bottom:6px;">`;
            html += `<div>${icon} <b>${escapeHtml(s.step)}</b> — ${escapeHtml(s.message)}</div>`;
            if (s.suggestion) html += `<div style="color:#fbbf24;font-size:11px;margin-left:20px;">💡 ${escapeHtml(s.suggestion)}</div>`;
            html += `</div>`;
          }
        }
        if (res.root_cause) {
          html += `<div style="margin-top:8px;padding:8px;background:#7f1d1d;border-radius:4px;color:#fca5a5;">🎯 根因: ${escapeHtml(res.root_cause)}</div>`;
        }
        if (res.suggestions && res.suggestions.length) {
          html += `<div style="margin-top:8px;color:#86efac;font-size:12px;">💡 修复建议:</div>`;
          for (const s of res.suggestions) {
            html += `<div style="color:#86efac;font-size:12px;margin-left:12px;">• ${escapeHtml(s)}</div>`;
          }
        }
        html += `</div>`;
        addMsg('ai', html);
      } catch(e) {
        addMsg('ai', `❌ 诊断请求失败: ${e.message}`);
      }
    }

          html += `<div style="display:flex;justify-content:space-between;align-items:center;padding:4px 8px;border-bottom:1px solid #334155;">`;
          html += `<div><span style="color:#94a3b8;">${time}</span> <span style="color:#64748b;font-size:10px;">${size}KB</span></div>`;

          html += `</div>`;

        html += `</div>`;
        addMsg('ai', html);
      } catch(e) {
        addMsg('ai', `❌ 查询快照失败: ${e.message}`);
      }
    }

    async function sendChatWithPreview(msg, context) {
      // 配置操作：先预览命令，用户确认后再执行
      const input = document.getElementById('chat-input');
      input.value = '';
      addMsg('system', '⏳ 正在生成命令预览...');

      try {
        const previewRes = await fetch('/api/chat', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({
          message: msg, context, preview_only: true
        })});
        const previewData = await previewRes.json();

        if (!previewData.success && !previewData.preview) {
          addMsg('ai', '❌ ' + (previewData.message || '预览失败'));
          return;
        }

        if (!previewData.preview || !previewData.planned_commands?.length) {
          // 没有工具调用，直接显示文本回复
          addMsg('user', msg);
          addMsg('ai', previewData.response || '无操作需要执行');
          return;
        }

        const cmds = previewData.planned_commands;
        // 在聊天区显示预览
        let previewHtml = '<div style="color:#f59e0b;font-weight:600;margin-bottom:6px;">⚠️ 配置命令预览</div>';
        for (const item of cmds) {
          previewHtml += `<div style="margin-bottom:6px;"><span style="color:#22d3ee;font-size:12px;font-weight:600;">📌 ${escapeHtml(item.device)}</span>`;
          if (item.commands?.length) {
            previewHtml += `<pre style="background:#0b1220;border:1px solid #334155;border-radius:4px;padding:6px 8px;margin-top:2px;font-size:11px;">${item.commands.map(c => escapeHtml(c)).join('\n')}</pre>`;
          }
          previewHtml += '</div>';
        }
        previewHtml += `<div style="display:flex;gap:6px;margin-top:8px;">`;
        previewHtml += `<button class="btn" style="background:#dc2626;color:#fff;" onclick="confirmChatExec(this,'${escapeHtml(JSON.stringify(cmds).replace(/'/g,"\\'"))}', '${escapeHtml(msg)}', '${escapeHtml(JSON.stringify(context))}')">确认执行</button>`;
        previewHtml += `<button class="btn" onclick="this.parentElement.parentElement.remove()">取消</button>`;
        previewHtml += `</div>`;
        addMsg('ai', previewHtml);
        // 移除预览提示
        const msgs = document.getElementById('messages');
        const sysMsg = msgs.querySelector('.msg-system:last-of-type');
        // 不移除，让它自然保留
      } catch(e) {
        addMsg('ai', '❌ 预览失败: ' + e.message);
      }
    }

    async function sendChat() {
      if (_chatBusy) return;
      const input = document.getElementById('chat-input');
      const msg = input.value.trim();
      if (!msg) return;

      // 从拓扑多选中获取目标设备
      const selected = Array.from(selectedIds).map(id => devices.find(d => d.id === id)).filter(Boolean);
      let context = {};
      if (selected.length > 1) {
        context.selected_devices = selected.map(d => d.remark || d.name);
        context.selected_device = selected.map(d => d.remark || d.name).join('、');
      } else if (selected.length === 1) {
        context.selected_device = selected[0].name || selected[0].remark;
      } else {
        const d = devices.find(x => x.id === selectedId);
        if (d) context.selected_device = d.remark || d.name;
      }

      // ====== 先尝试快速模板匹配（不走 LLM，秒回）======
      const selDev = devices.find(x => x.id === selectedId);
      if (selDev) {
        try {
          const qr = await fetch('/api/quick-config', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({
            message: msg, device: selDev?.name||selDev?.remark, mode: 'preview'
          })});
          const qd = await qr.json();
          if (qd.matched) {
            input.value = '';
            if (qd.query) {
              addMsg('user', msg);
              const parts = [];
              for (const r of (qd.results||[])) {
                if (r.success && r.results) {
                  for (const x of r.results) { parts.push(`命令: ${x.command}\n${x.output?.slice(0,2000)||''}`); }
                } else if (!r.success) { parts.push(`❌ ${r.error||'失败'}`); }
              }
              addMsg('ai', parts.join('\n---\n') || '查询完成');
              if(selDev) collectDeviceFacts(selDev?.name||selDev?.remark);
            } else if (qd.preview) {
              await sendChatWithPreview(msg, context);
            }
            return;
          }
        } catch(e) { /* 快速模板失败，fallback 到 LLM */ }
      }

      // 配置操作走预览确认流程（不再用 confirm 弹窗）
      const cfgWords = ['配置','创建','删除','修改','关闭','shutdown','undo','撤销'];
      const isConfigOp = cfgWords.some(k => msg.includes(k)) && !msg.includes('查询') && !msg.includes('查看') && !msg.includes('display') && !msg.includes('show');

      if (isConfigOp) {
        await sendChatWithPreview(msg, context);
        return;
      }

      addMsg('user', msg);
      input.value = '';
      _chatBusy = true;
      _chatAbort = new AbortController();
      document.getElementById('btn-stop').style.display = '';
      document.getElementById('btn-send').style.display = 'none';
      
      // 思考中占位
      const thinkingHtml = '<div class="chat-msg ai" id="thinking-placeholder"><span style="color:#67e8f9">⏳ 思考中...</span></div>';
      document.getElementById('chat-messages').insertAdjacentHTML('beforeend', thinkingHtml);
      
      try {
        const r = await fetch('/api/chat', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ message: msg, context }),
          signal: _chatAbort.signal
        });
        const res = await r.json();
        const tp = document.getElementById('thinking-placeholder');
        if (tp) tp.remove();

        // 如果LLM返回了命令预览，走确认流程
        if (res.preview && res.planned_commands?.length) {
          const cmds = res.planned_commands;
          let previewHtml = '<div style="color:#f59e0b;font-weight:600;margin-bottom:6px;">⚠️ 命令预览</div>';
          cmds.forEach(c => {
            previewHtml += `<div style="margin-bottom:6px;"><span style="color:#22d3ee;font-size:12px;font-weight:600;">${c.device}</span>`;
            previewHtml += `<pre style="background:#0b1220;border:1px solid #334155;border-radius:4px;padding:8px;font-size:12px;overflow-x:auto;">${c.commands.join('\n')}</pre>`;
            previewHtml += '</div>';
          });
          previewHtml += `<div style="display:flex;gap:6px;margin-top:8px;">`;
          const cmdKey = '_pcmd_' + Date.now();
          window[cmdKey] = { cmds, msg, context };
          previewHtml += `<button class="btn" style="background:#16a34a;color:#fff;" onclick="confirmChatExec(this, window['${cmdKey}'].cmds, window['${cmdKey}'].msg, window['${cmdKey}'].context); delete window['${cmdKey}']">确认执行</button>`;
          previewHtml += `<button class="btn" onclick="this.parentElement.parentElement.remove()">取消</button>`;
          previewHtml += `</div>`;
          // 先显示LLM的文本回复
          if (res.response) addMsg('ai', res.response);
          addMsg('ai', previewHtml);
        } else {
          addMsg('ai', (res.executed ? '✅ ' : '') + (res.response || '无响应'));
        }
      } catch (e) {
        const tp = document.getElementById('thinking-placeholder');
        if (tp) tp.remove();
        if (e.name === 'AbortError') {
          addMsg('system', '⏹ 已停止');
        } else {
          addMsg('ai', '❌ 请求失败');
        }
      } finally {
        _chatBusy = false;
        _chatAbort = null;
        document.getElementById('btn-stop').style.display = 'none';
        document.getElementById('btn-send').style.display = '';
      }
    }

    // ===== 统一执行 =====
    function execForm(type, dev) {
      const v = id => { const e = document.getElementById(id); return e ? e.value.trim() : ''; };
      const d = devices.find(x => x.id === selectedId);
      const vendor = d?.vendor || 'unknown';
      const isHua = vendor==='huawei'||vendor==='h3c'||vendor==='ruijie';
      const isCis = vendor==='cisco'||vendor==='cisco_nxos'||vendor==='arista'||vendor==='juniper';

      let commands = []; // 直接生成命令列表
      let msg = '';      // 用户可见描述

      switch(type) {
        case 'vlan': {
          const act=v('cfg-vlan-action'), vid=v('cfg-vlan-id'), nm=v('cfg-vlan-name');
          if(!vid){alert('填 VLAN ID');return;}
          if(act==='create') {
            msg = `创建 VLAN ${vid}${nm?' 名称 '+nm:''}`;
            if(isHua) commands = [`vlan ${vid}`, nm?`name ${nm}`:`description VLAN_${vid}`, 'quit'];
            else if(isCis) commands = [`vlan ${vid}`, nm?`name ${nm}`:`name VLAN_${vid}`, 'exit'];
          } else {
            msg = `删除 VLAN ${vid}`;
            if(isHua) commands = [`undo vlan ${vid}`];
            else if(isCis) commands = [`no vlan ${vid}`];
          }
          break;
        }
        case 'interface': {
          const port=v('cfg-if-port'), act=v('cfg-if-action'), val=v('cfg-if-val');
          if(!port){alert('填接口');return;}
          if(act==='shutdown') {
            msg=`关闭接口 ${port}`;
            commands = [`interface ${port}`, 'shutdown', isHua?'quit':'exit'];
          } else if(act==='noshutdown') {
            msg=`开启接口 ${port}`;
            commands = [`interface ${port}`, isHua?'undo shutdown':'no shutdown', isHua?'quit':'exit'];
          } else if(act==='desc') {
            msg=`设置接口 ${port} 描述 "${val}"`;
            commands = [`interface ${port}`, `description ${val}`, isHua?'quit':'exit'];
          } else if(act==='speed') {
            msg=`设置接口 ${port} 速率 ${val||'auto'}`;
            commands = [`interface ${port}`, `speed ${val||'auto'}`, isHua?'quit':'exit'];
          } else if(act==='duplex') {
            msg=`设置接口 ${port} 双工 ${val||'auto'}`;
            commands = [`interface ${port}`, `duplex ${val||'auto'}`, isHua?'quit':'exit'];
          }
          break;
        }
        case 'interface-batch': {
          const range=v('cfg-if-range'), act=v('cfg-if-batch-action'), desc=v('cfg-if-batch-desc');
          if(!range){alert('填接口范围');return;}
          // 华为用 interface range，思科用 interface range
          const ifCmd = isHua ? `interface range ${range}` : `interface range ${range}`;
          if(act==='shutdown') { msg=`批量关闭 ${range}`; commands=[ifCmd,'shutdown','quit']; }
          else if(act==='noshutdown') { msg=`批量开启 ${range}`; commands=[ifCmd,isHua?'undo shutdown':'no shutdown','quit']; }
          else if(act==='desc') { msg=`批量设置描述`; commands=[ifCmd,`description ${desc}`,'quit']; }
          break;
        }
        case 'portmode': {
          const port=v('cfg-pm-port'), mode=v('cfg-pm-mode'), vlan=v('cfg-pm-vlan'), allow=v('cfg-pm-allow');
          if(!port){alert('填接口');return;}
          if(mode==='access') {
            msg=`接口 ${port} → Access VLAN ${vlan||'?'}`;
            if(isHua) commands=[`interface ${port}`,'port link-type access',vlan?`port access vlan ${vlan}`:'',isHua?'quit':'exit'].filter(Boolean);
            else if(isCis) commands=[`interface ${port}`,'switchport mode access',vlan?`switchport access vlan ${vlan}`:'',isCis?'exit':'exit'].filter(Boolean);
          } else {
            msg=`接口 ${port} → Trunk` + (allow?` 允许VLAN ${allow}`:'');
            if(isHua) commands=[`interface ${port}`,'port link-type trunk',allow?`port trunk permit vlan ${allow}`:'port trunk permit vlan all','quit'].filter(Boolean);
            else if(isCis) commands=[`interface ${port}`,'switchport mode trunk',allow?`switchport trunk allowed vlan ${allow}`:'','exit'].filter(Boolean);
          }
          break;
        }
        case 'l3-if': {
          const ft=v('cfg-l3-iftype'), iface=v('cfg-l3-iface'), ip=v('cfg-l3-ip');
          if(!iface||!ip){alert('填接口和 IP');return;}
          msg=`配置${ft==='svi'?'VLAN接口':ft==='loopback'?'Loopback':'物理接口'} ${iface}，IP ${ip}`;
          if(isHua) commands=[`interface ${iface}`,`ip address ${ip}`,'quit'];
          else if(isCis) commands=[`interface ${iface}`,`ip address ${ip}`,'exit'];
          break;
        }
        case 'l3-route': {
          const dst=v('cfg-l3-rt-dst'), nh=v('cfg-l3-rt-nh');
          if(!dst||!nh){alert('填目标和下一跳');return;}
          msg=`静态路由 ${dst} → ${nh}`;
          commands = isHua ? [`ip route-static ${dst} ${nh}`] : [`ip route ${dst} ${nh}`];
          break;
        }
        case 'l3-ospf': {
          const id=v('cfg-l3-ospf-id')||'1', net=v('cfg-l3-ospf-net');
          if(!net){alert('填网段');return;}
          msg=`OSPF 进程 ${id} 宣告 ${net}`;
          if(isHua) commands=[`ospf ${id}`,`network ${net}`,'quit'];
          else if(isCis) commands=[`router ospf ${id}`,`network ${net} area 0`];
          break;
        }
        case 'ping': {
          const ip=v('cfg-dg-ping'), c=v('cfg-dg-ping-c')||'5';
          if(!ip){alert('填目标 IP');return;}
          msg=`ping ${ip}`;
          commands = [isHua?`ping ${ip} count ${c}`:`ping ${ip} repeat ${c}`];
          break;
        }
        case 'traceroute': {
          const ip=v('cfg-dg-trace');
          if(!ip){alert('填目标');return;}
          msg=`traceroute ${ip}`;
          commands = [isHua?`tracert ${ip}`:`traceroute ${ip}`];
          break;
        }
        default: {
          // 其余复杂表单（BGP/DHCP/ACL/NAT/zone/sys）还是走 LLM
          // 回退到原来的自然语言方式
          return execFormLegacy(type, dev);
        }
      }

      if(!commands.length && !msg) return;
      const resultId = `cfg-${type}-result`;

      // 判断是查询还是配置
      const isQuery = commands.every(c => c.toLowerCase().startsWith('display') || c.toLowerCase().startsWith('show') || c.toLowerCase().startsWith('ping') || c.toLowerCase().startsWith('trac'));
      
      if(isQuery) {
        // 查询直接执行
        doExecDirect(dev, msg, commands, resultId);
      } else {
        // 配置走预览确认
        doExecDirectWithPreview(dev, msg, commands, resultId);
      }
    }

    async function doExecDirect(dev, msg, commands, resultId) {

      if(el){el.style.display='block';el.className='config-result';el.textContent='⏳ 执行中...';}

          device: d?.name||d?.remark, mode:'execute', commands:[{device:d?.name||d?.remark, commands}]
        })});
        const res = await r.json();
        const output = (res.results||[]).map(r => {
          if(r.success && r.results) return r.results.map(x=>`命令: ${x.command}\n${x.output?.slice(0,2000)||''}`).join('\n---\n');

        if(el){el.className='config-result error';el.textContent='请求失败';}
      }
    }

    function execFormLegacy(type, dev) {
      // 复杂表单回退走 LLM
      const v = id => { const e = document.getElementById(id); return e ? e.value.trim() : ''; };
      let msg = '';
      switch(type) {
        case 'portspecial': {
          const sp=v('cfg-pm-special'),port=v('cfg-pm-sp-port'),grp=v('cfg-pm-sp-group');
          if(!port||!grp){alert('填接口和组号');return;}
          const map={'agg-lacp':'配置 LACP 端口聚合','agg-static':'配置静态端口聚合','mirror-src':'配置镜像源端口','mirror-dst':'配置镜像目的端口'};
          msg=`在 ${dev} 上${map[sp]||sp}，接口 ${port}，组号 ${grp}`; break;
        }
        case 'l3-bgp': {
          const as=v('cfg-l3-bgp-as'),nbr=v('cfg-l3-bgp-nbr');
          if(!as){alert('填 AS 号');return;}
          msg=`在 ${dev} 上配置 BGP AS ${as}`; if(nbr) msg+=`，邻居 ${nbr}`; break;
        }
        case 'l3-dhcp': {
          const pool=v('cfg-l3-dhcp-pool'),net=v('cfg-l3-dhcp-net');
          if(!pool||!net){alert('填地址池和网段');return;}
          msg=`在 ${dev} 上配置 DHCP 地址池 ${pool}，${net}`; break;
        }
        case 'acl': {
          const act=v('cfg-sec-acl-act'),rid=v('cfg-sec-acl-id'),perm=v('cfg-sec-acl-perm'),src=v('cfg-sec-acl-src'),dst=v('cfg-sec-acl-dst'),proto=v('cfg-sec-acl-proto'),iface=v('cfg-sec-acl-if');
          if(act==='delete'){if(!rid){alert('填规则号');return;} msg=`在 ${dev} 上删除 ACL 规则 ${rid}`; break;}
          if(!src&&!dst){alert('填源或目标');return;}
          msg=`在 ${dev} 上创建 ACL 规则 ${rid||'自动'}，${perm} ${src||'any'} → ${dst||'any'}`;
          if(proto) msg+=` ${proto}`; if(iface) msg+=`，应用到 ${iface}`; break;
        }
        case 'nat': {
          const nt=v('cfg-sec-nat-type'),src=v('cfg-sec-nat-src'),ext=v('cfg-sec-nat-ext'),port=v('cfg-sec-nat-port');
          if(!src||!ext){alert('填内部地址和外部信息');return;}
          if(nt==='snat') msg=`在 ${dev} 上配置源 NAT，${src} 经 ${ext} 转换`;
          else msg=`在 ${dev} 上配置目的 NAT，外部 ${ext}→${src}${port?' 端口 '+port:''}`; break;
        }
        case 'zone': {
          const act=v('cfg-sec-zone-act'),nm=v('cfg-sec-zone-name'),val=v('cfg-sec-zone-val');
          if(!nm){alert('填域名');return;}
          const map={create:'创建安全域',addif:'将 '+val+' 加入安全域',policy:'配置域间策略'};
          msg=`在 ${dev} 上${map[act]||act} ${nm}`; if(act==='policy') msg+=`：${val}`; break;
        }
        case 'sys-name': { const nm=v('cfg-sys-name'); if(!nm){alert('填名称');return;} msg=`在 ${dev} 上设置设备名称为 ${nm}`; break; }
        case 'sys-mgmt': { const mg=v('cfg-sys-mgmt'); if(!mg){alert('填管理 IP');return;} msg=`在 ${dev} 上配置管理地址 ${mg}`; break; }
        case 'sys-sec': {
          const act=v('cfg-sys-sec-act'),val=v('cfg-sys-sec-val');
          const map={passwd:'设置 Console 密码',ssh:'启用 SSH 远程管理',telnet:'启用 Telnet 远程管理','acl-remote':'限制远程访问 IP 为 '+val};
          msg=`在 ${dev} 上${map[act]||act}${act==='passwd'?' 密码 '+val:''}`; break;
        }
      }
      if(!msg) return;
      const resultId = `cfg-${type}-result`;
      doExec(dev, msg, resultId);
    }

    function execBatchVlan(dev) {
      const text = document.getElementById('cfg-vlan-batch').value.trim();
      if(!text){alert('填 VLAN 列表');return;}
      const lines = text.split('\n').filter(l=>l.trim());

    async function doExec(dev, msg, resultId) {
      // 判断是否为配置操作（需要预览确认）
      const cfgWords = ['配置','创建','删除','关闭','shutdown','undo','撤销','开启','开启接口','关闭接口','Trunk','Access','静态路由','OSPF','BGP','NAT','DHCP','ACL','安全域','SSH','SNMP','端口聚合','镜像','描述'];
      const isConfigOp = cfgWords.some(k => msg.includes(k));

      if (isConfigOp) {
        // 配置操作走预览确认流程
        await doExecWithPreview(dev, msg, resultId);
        return;
      }

      // 查询操作直接执行
      addMsg('user', msg);
      const el = document.getElementById(resultId);
      if(el){el.style.display='block';el.className='config-result';el.textContent='⏳ 执行中...';}
      const d = devices.find(x => x.id === selectedId);
      try {
        const r = await fetch('/api/chat', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({message:msg,context:{selected_device:d?.name||d?.remark}})});
        const res = await r.json();
        if(el){
          el.className = res.success ? 'config-result success' : 'config-result error';
          el.textContent = res.response || res.message || (res.success?'成功':'失败');
        }
        addMsg('ai', (res.success?'✅ ':'❌ ')+(res.response||res.message||''));
        if(res.success) collectDeviceFacts(d?.name||d?.remark);
      } catch(e) {
        if(el){el.className='config-result error';el.textContent='请求失败';}
        addMsg('ai','❌ 请求失败: '+e.message);
      }
    }

    async function doExecWithPreview(dev, msg, resultId) {
      const el = document.getElementById(resultId);
      if(el){el.style.display='block';el.className='config-result';el.textContent='⏳ 生成命令预览...';}
      const d = devices.find(x => x.id === selectedId);
      try {
        // 第一步：预览，只生成命令不执行
        const previewRes = await fetch('/api/chat', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({
          message: msg,
          context: {selected_device: d?.name||d?.remark},
          preview_only: true
        })});
        const previewData = await previewRes.json();

        if (!previewData.success && !previewData.preview) {
          if(el){el.className='config-result error';el.textContent=previewData.message||'预览失败';}
          return;
        }

        if (!previewData.preview || !previewData.planned_commands?.length) {
          // LLM 没有生成工具调用（可能是纯文本回复），直接显示
          addMsg('user', msg);
          addMsg('ai', previewData.response || '无操作需要执行');
          if(el){el.style.display='none';}
          return;
        }

        // 第二步：显示确认弹窗
        const cmds = previewData.planned_commands;
        let cmdListHtml = '';
        for (const item of cmds) {
          cmdListHtml += `<div style="margin-bottom:8px;">`;
          cmdListHtml += `<div style="color:#22d3ee;font-size:12px;font-weight:600;">📌 设备: ${item.device}</div>`;
          if (item.commands?.length) {
            cmdListHtml += `<div style="background:#0b1220;border:1px solid #334155;border-radius:4px;padding:6px 8px;margin-top:2px;font-family:monospace;font-size:11px;">`;
            cmdListHtml += item.commands.map(c => `<div style="color:#e2e8f0;padding:1px 0;">${escapeHtml(c)}</div>`).join('');
            cmdListHtml += `</div>`;
          } else {
            cmdListHtml += `<div style="color:#6b7280;font-size:11px;">（无具体命令）</div>`;
          }
          cmdListHtml += `</div>`;
        }

        // 用内联确认区域代替 confirm() 弹窗
        let confirmBox = document.getElementById('cfg-confirm-box');
        if (!confirmBox) {
          confirmBox = document.createElement('div');
          confirmBox.id = 'cfg-confirm-box';
          confirmBox.style.cssText = 'position:sticky;bottom:0;background:#1a1030;border:1px solid #f59e0b;border-radius:8px;padding:12px;margin-top:8px;z-index:10;';
          const panel = document.getElementById('config-panel');
          if (panel) panel.appendChild(confirmBox);
        }
        confirmBox.innerHTML = `
          <div style="color:#f59e0b;font-size:13px;font-weight:600;margin-bottom:6px;">⚠️ 配置命令预览 — 请确认后执行</div>
          ${cmdListHtml}
          <div style="display:flex;gap:8px;margin-top:10px;">
            <button id="btn-confirm-exec" style="flex:1;padding:8px;background:#dc2626;color:#fff;border:none;border-radius:6px;cursor:pointer;font-weight:600;">确认执行</button>
            <button id="btn-cancel-exec" style="flex:1;padding:8px;background:#334155;color:#e5e7eb;border:none;border-radius:6px;cursor:pointer;">取消</button>
          </div>
        `;
        confirmBox.style.display = 'block';

        if(el){el.style.display='none';}

        // 确认按钮
        document.getElementById('btn-confirm-exec').onclick = async () => {
          confirmBox.style.display = 'none';
          addMsg('user', msg);
          if(el){el.style.display='block';el.className='config-result';el.textContent='⏳ 执行中...';}
          try {
            // 第三步：确认后直接执行命令
            const execRes = await fetch('/api/chat', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({
              message: msg,
              context: {selected_device: d?.name||d?.remark},
              confirmed_commands: cmds
            })});
            const execData = await execRes.json();
            if(el){
              el.className = execData.success ? 'config-result success' : 'config-result error';
              el.textContent = execData.response || execData.message || (execData.success?'成功':'失败');
            }
            addMsg('ai', (execData.success?'✅ ':'❌ ')+(execData.response||execData.message||''));
            if(execData.success) collectDeviceFacts(d?.name||d?.remark);
          } catch(e) {
            if(el){el.className='config-result error';el.textContent='执行失败';}
            addMsg('ai','❌ 执行失败: '+e.message);
          }
        };

        // 取消按钮
        document.getElementById('btn-cancel-exec').onclick = () => {
          confirmBox.style.display = 'none';
          if(el){el.style.display='none';}
          addMsg('system', '⏹ 已取消配置操作');
        };
      } catch(e) {
        if(el){el.className='config-result error';el.textContent='预览失败';}
        addMsg('ai','❌ 预览请求失败: '+e.message);
      }
    }

    async function quickQuery(device, msg) {
      const el = document.getElementById('cfg-query-result') || document.getElementById('cfg-dg-tool-result');

      const d = devices.find(x => x.id === selectedId);
      const devName = d?.remark || d?.name || device;
      try {
        // 先尝试快速模板（秒回）
        let res = null;
        try {
          const qr = await fetch('/api/quick-config', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({device:devName,mode:'preview',message:msg})});
          const qd = await qr.json();
          if(qd.matched && qd.results) {
            const parts = [];
            for(const r of (qd.results||[])) {
              if(r.success && r.results) { for(const x of r.results) { parts.push(`命令: ${x.command}\n${x.output?.slice(0,2000)||''}`); } }
              else if(!r.success) { parts.push(`❌ ${r.error||'失败'}`); }
            }
            res = {success:true, response: parts.join('\n---\n')||'查询完成', _fast:true};
          } else if(qd.commands && qd.preview) {
            // 模板匹配返回命令列表，走确认流程
            const planned = [{device: qd.device||devName, commands: qd.commands}];
            let previewHtml = '<div style="color:#f59e0b;font-weight:600;margin-bottom:6px;">⚠️ 配置命令预览</div>';
            planned.forEach(c => {
              previewHtml += `<div style="margin-bottom:6px;"><span style="color:#22d3ee;font-size:12px;font-weight:600;">${c.device}</span>`;
              previewHtml += `<pre style="background:#0b1220;border:1px solid #334155;border-radius:4px;padding:8px;font-size:12px;overflow-x:auto;">${c.commands.join('\n')}</pre>`;
              previewHtml += '</div>';
            });
            previewHtml += `<div style="display:flex;gap:6px;margin-top:8px;">`;
            const cmdKey = '_pcmd_' + Date.now();
            window[cmdKey] = { cmds: planned, msg, context: {selected_device: devName} };
            previewHtml += `<button class="btn" style="background:#16a34a;color:#fff;" onclick="confirmChatExec(this, window['${cmdKey}'].cmds, window['${cmdKey}'].msg, window['${cmdKey}'].context); delete window['${cmdKey}']">确认执行</button>`;
            previewHtml += `<button class="btn" onclick="this.parentElement.parentElement.remove()">取消</button>`;
            previewHtml += `</div>`;
            addMsg('ai', previewHtml);
            res = {success:true, response: '配置命令已生成', _fast:true};
          }
        } catch(e) { /* fallback to LLM */ }
        // 快速模板没匹配到，走LLM
        if(!res) {
          const r = await fetch('/api/chat', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({message:msg,context:{selected_device:devName}})});
          res = await r.json();
        }
        if(el){el.className=res.success?'config-result success':'config-result error';el.textContent=res.response||res.message||'';}
        addMsg('ai',(res.success?'✅ ':'❌ ')+(res.response||res.message||''));
      } catch(e) { if(el){el.className='config-result error';el.textContent='失败';} }
    }

    async function collectDeviceFacts(deviceName) {
      if(!deviceName) return;
      try {
        await fetch('/api/device/collect',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({device:deviceName})});
        await loadDevices(); buildTopologyFromDevices();
        const nodes=topologyState.nodes||[]; for(const n of nodes){const dev=devices.find(x=>x.id===n.id);if(dev?.facts) n.facts=dev.facts;}
        renderTopology();
      } catch(e){console.error('采集失败:',e);}
    }

    async function refreshAllFacts() {
      if(!devices.length){addMsg('system','暂无设备');return;}
      addMsg('system','⏳ 采集中...');
      for(const d of devices){if(d.remark||d.ip) await collectDeviceFacts(d.remark||d.ip);}
      addMsg('system','✅ 采集完成');
    }

    // 拓扑实时刷新
    let _topoTimer = null;
    let _discovering = false;  // 防止并发discover
    async function startTopoRefresh(sec) {
      stopTopoRefresh();
      if(sec > 0) {
        _topoTimer = setInterval(async () => {
          if (_discovering) return;  // 上一次还没结束，跳过
          try {
            await discoverTopology();
          } catch(e) {}
        }, sec * 1000);
      }
    }
    function stopTopoRefresh() { if(_topoTimer){clearInterval(_topoTimer);_topoTimer=null;} }

    async function deleteCurrentDevice() {
      if (!confirm('确认删除该设备？')) return;
      try {
        const r = await fetch('/api/device/delete', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({id:editingDeviceId})});
        const res = await r.json();
        if (res.success) {
          closeModal('modal-device');
          selectedId = null;
          editingDeviceId = null;
          await loadDevices();
          buildTopologyFromDevices();
          await syncTopologyState();
          renderAll();
          addMsg('system','✅ 设备已删除');
        } else {
          alert(res.message||'删除失败');
        }
      } catch(e) { alert('删除失败: ' + e.message); }
    }

      try {
        const r = await fetch('/api/audit/logs?limit=50');

        if (d.logs && d.logs.length > 0) {
          container.innerHTML = d.logs.map(log => {
            const cmds = log.details?.commands || [];
            const cmdsHtml = cmds.length ? `<div style="color:#94a3b8;font-size:10px;margin-top:2px;font-family:monospace;">${cmds.map(c=>escapeHtml(c)).join(' → ')}</div>` : '';
            const inputHtml = log.details?.input ? `<div style="color:#64748b;font-size:10px;margin-top:1px;">${escapeHtml(log.details.input)}</div>` : '';
            return `
            <div class="audit-item">
              <div class="audit-time">${log.timestamp || ''}</div>
              <div><span class="audit-action">${log.action || ''}</span> <span class="audit-target">${log.target || ''}</span></div>

        }
      } catch (e) {
        container.innerHTML = '<div style="color:#ef4444;font-size:12px;">加载失败</div>';
      }
    }
